# =============================================================================
# capstone_project — thư viện cần cài để build + chạy toàn bộ package (trừ da1)
# Nền: Ubuntu 22.04 + ROS 2 Humble (đã cài /opt/ros/humble, ví dụ ros-humble-desktop
#      hoặc ros-humble-ros-base). Mọi thứ dưới đây đều cài bằng apt, KHÔNG cần pip.
#
# Cài tất cả (bỏ qua dòng chú thích):
#   sudo apt update
#   sudo apt install -y $(sed 's/#.*//' requirements.txt | xargs)
#
# Sau đó (tuỳ chọn, kiểm tra lại bằng rosdep):
#   sudo rosdep init        # chỉ lần đầu
#   rosdep update
#   cd ~/capstone_ws && rosdep install --from-paths capstone_project \
#       --ignore-src -r -y --skip-keys "da1"
#
# Build:
#   source /opt/ros/humble/setup.bash
#   cd ~/capstone_ws && colcon build --symlink-install --packages-skip da1
#   source install/setup.bash
#
# Phiên bản đã dùng trên máy dev (2026-10): Nav2/DWB 1.1.20, slam_toolbox 2.6.10,
# robot_localization 3.5.4, librealsense2 2.57.7, realsense2_camera 4.57.7, libmodbus 3.1.6.
# =============================================================================
